### Performance Testing Strategy
```python
# Example performance test pattern
def test_context_builder_cache_hit_performance():
    builder = ContextBuilder()
    original_context = {"docs_context_summary": mock_summary}

    start = time.perf_counter()
    for _ in range(100):
        context = builder.build(..., original_context=original_context)
    elapsed = time.perf_counter() - start

    avg_time = elapsed / 100
    assert avg_time < 0.010  # 10ms threshold
```

### Mocking Disk Access
To verify that cache hits avoid disk reads:
```python
# Mock _resolve_reference_docs to detect calls
with patch.object(builder, '_resolve_reference_docs') as mock_resolve:
    context = builder.build(..., original_context={"docs_context_summary": mock_summary})
    mock_resolve.assert_not_called()  # Should not hit disk
```

### Checkpoint Testing
```python
# Simulate checkpoint save/restore
async def test_pipeline_resume_restores_docs_context():
    # Start pipeline
    state = create_initial_state(pipeline_id, subject_context, docs_context_summary=mock_summary)
    # ... execute some nodes ...

    # Simulate checkpoint (LangGraph handles this automatically)
    checkpoint = graph.get_state(config)

    # Resume from checkpoint

    assert resumed_state["docs_context_summary"] == mock_summary
```
