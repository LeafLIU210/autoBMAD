```python
class CreateDeliverableTool(CallableTool2):
    async def __call__(self, params: CreateDeliverableParams) -> ToolReturnValue:
        filename = f"{params.title.lower().replace(' ', '-')}.md"
        file_path = Path.cwd() / filename  # Relative to SDK work_dir

        try:
            async with aiofiles.open(file_path, "w", encoding="utf-8") as f:
                await f.write(params.content)
            return ToolOk(output=f"Deliverable '{params.title}' saved to {file_path}")
        except Exception as exc:
            return ToolError(output="", message=str(exc), brief="Failed to write deliverable")
```
